cmake_minimum_required(VERSION 3.24)
project(WindowFocusGuard VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Requires Windows x64 MSVC")
endif()
set(SKSE_SDK_ROOT "" CACHE PATH "External pinned SKSE source checkout")
file(SHA256 "${SKSE_SDK_ROOT}/skse64/PluginAPI.h" sdk_hash)
if(NOT sdk_hash STREQUAL "0f238cc870c726a91befcf96f1c45db0019e223242434b8300b973ab2b0a3c95")
  message(FATAL_ERROR "Unexpected SKSE ABI header; review before updating pin")
endif()
add_library(WindowFocusGuard SHARED WindowFocusGuard.cpp)
target_include_directories(WindowFocusGuard PRIVATE "${SKSE_SDK_ROOT}")
target_link_libraries(WindowFocusGuard PRIVATE user32 shell32 ole32)
add_executable(FocusPolicyTests FocusPolicyTests.cpp)
add_executable(ClipLeaseTests ClipLeaseTests.cpp)
foreach(target WindowFocusGuard FocusPolicyTests ClipLeaseTests)
  target_compile_features(${target} PRIVATE cxx_std_20)
  target_compile_options(${target} PRIVATE /W4 /WX /permissive- /Brepro)
  target_link_options(${target} PRIVATE /Brepro)
  set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endforeach()
enable_testing()
add_test(NAME foreground_policy COMMAND FocusPolicyTests)
add_test(NAME clip_ownership COMMAND ClipLeaseTests)
